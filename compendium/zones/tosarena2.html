SomethingTryGo(what = $, row = $, col = $)

{
     local i;

     for i in plExits
     {
        if (row = first(i) and col = nth(i,2)) and nth(i,3) = RID_TOS_ARENA
        {
           if send(send(SYS,@findroombynum,#num=RID_TOS_ARENA),@isLocked)
             {
                send(what,@msgsenduser,#message_rsc=TosArena2_locked);
                return TRUE;
             }
        }
     }
     propagate;
  }
