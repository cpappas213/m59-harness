ResetTimer()

{
      ptReset = $;

      if piPosition = DOOR_OPEN
      {
         Send(self,@SomethingWaveRoom,#wave_rsc=templekra_sound);
         send(self,@SetSector,#sector=SECTOR_DOOR,#animation=ANIMATE_CEILING_LIFT,
              #height=92,#speed=16);
         piPosition = DOOR_CLOSED;
      }
     
      return;
   }
