EnterRentableRoom(who = $)

Sees if we are in the proper location to enter a rentable room in this inn.

{
      local iRow, iCol;

      iRow = send(who,@GetRow);
      iCol = send(who,@GetCol);
      
      
      if ( iRow = 4
           AND (iCol = 12
                 OR iCol = 13
                 OR iCol = 17) )
         OR ( iCol = 5
              AND (iRow = 1
                   OR iRow = 2
                   OR iRow = 7) )
         OR ( iCol = 7
              AND (iRow = 2
                   OR iRow = 3
                   OR iRow = 11) )
         OR ( iRow = 11
              AND iCol = 6 )
      {
         return TRUE;
      }

      return FALSE;
   }
