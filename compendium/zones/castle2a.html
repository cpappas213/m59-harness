SomethingTryGo(what = $, row = $, col = $)

{
      local oLockedRoom;

      oLockedRoom = Send(SYS,@FindRoomByNum,#num=RID_BAZMANS_ROOM);
   
      if row = 27 AND (col = 15 OR col = 16)
      {
         if Send(oLockedRoom,@CheckLock) = 1
         { 
            Send(what,@TeleportTo,#rid=RID_FORGOTTEN_TOO);

            return TRUE;
         }
      }

      propagate;
   }
